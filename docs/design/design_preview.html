SELECT ROUND(AVG(j.salary_avg),1) AS avg_k,
       COUNT(*) AS n
FROM job_intel j
JOIN job_skills s ON s.job_id = j.id
WHERE j.city = '广州'
  AND j.experience = '经验不限'
  AND s.skill = 'RAG';
